69void dessiner_entite(int type_entite, int position_x, int position_y,int frame,int direction) {
70 int nb_image=0;

127SDL_Rect obtenir_hitbox_entite(int type_entite, int position_x, int position_y) {
128 int nb_image = 0;
129 int nb_ligne = 0;
130 float zoom = 1.0;
131 SDL_Texture* texture_actuelle = NULL;
132
133 //On récupère les mêmes propriétés que pour le dessin selon l'entite (le sprite)
134 if (type_entite == 1) { // Chèvre
135 zoom = 1.5;
136 texture_actuelle = texture_chevre;
137 nb_image = 4;
138 nb_ligne = 6;
139 }
140 else if (type_entite == 2) { // Fermier
141 zoom = 1.0;
142 texture_actuelle = texture_fermier;
143 nb_image = 9;
144 nb_ligne = 4;
145 }
146 else if (type_entite == 3) { // Chevreau
147 zoom = 1.0;
148 texture_actuelle = texture_chevreau;
149 nb_image = 4;
150 nb_ligne = 5;
151 }
152 else if (type_entite == 4) { // Loup
153 zoom = 0.5;
154 texture_actuelle = texture_wolf;
155 nb_image = 4;
156 nb_ligne = 6;
157 }
158
159 //Calcul de la taille globale du sprite
160 int texture_w = 0, texture_h = 0;
161 SDL_QueryTexture(texture_actuelle, NULL, NULL, &texture_w, &texture_h);
162
163 int offset_x = texture_w / nb_image;
164 int offset_y = texture_h / nb_ligne;
165
166 SDL_Rect hitbox;
167 hitbox.x = position_x;
168 hitbox.y = position_y;
169 hitbox.w = offset_x * zoom;
170 hitbox.h = offset_y * zoom;
171
172 // 3. Personnalisation de la taille de la hitbox on peut changer manuellement les marges
173 // On réduit le rectangle pour coller au corps réel de l'entite
174 int marge_x = 0;
175 int marge_y = 0;
176
177 if (type_entite == 1) { // Chèvre : on retire par exemple 15px sur les côtés
178 marge_x = 15;
179 marge_y = 10;
180 }
181 else if (type_entite == 2) { // Fermier : souvent plus vertical
182 marge_x = 20;
183 marge_y = 5;
184 }
185 else if (type_entite == 4) { // Loup
186 marge_x = 10;
187 marge_y = 8;
188 }
189
190 // Application des marges pour recentrer la hitbox
191 hitbox.x += marge_x;
192 hitbox.y += marge_y;
193 hitbox.w -= (marge_x * 2);
194 hitbox.h -= (marge_y * 2);
195
196 return hitbox;
197}
