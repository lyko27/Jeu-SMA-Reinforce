28int choisir_action_softmax(float *tab_interet, int nb_actions)
29{
30 float max_interet = tab_interet[0];
31 for (int i = 1; i < nb_actions;
32 i++)
33 { // ici on l'action avec le meilleur interet
34 if (tab_interet[i] > max_interet)
35 {
36 max_interet = tab_interet[i];
37 }
38 }
39
40 // on calcule exp(interet de chaque action) et on met ca dans le tableau de
41 // probabilite
42 float somme_exp = 0.0f;
43 float probabilites[nb_actions];
44
45 for (int i = 0; i < nb_actions; i++)
46 {
47 probabilites[i] = exp(tab_interet[i] - max_interet);
48 somme_exp +=
49 probabilites[i]; // on calcul la somme des probabilites en meme temps
50 }
51
52 float tirage = (float)rand() /
53 (float)RAND_MAX; // le tirage simule le de dont on a parle pour
54 // le cas d un environnement stochastique
55 float somme_cumulee = 0.0f;
56 for (int i = 0; i < nb_actions; i++)
57 {
58 somme_cumulee +=
59 (probabilites[i] / somme_exp); // on normalise la proba pour que proba
60 // soit comprise entre 0 et 1
61 if (tirage <= somme_cumulee)
62 {
63 return i;
64 }
65 }
66 return 0; // securite
67}
